Mémo — Manipuler les tableaux en JavaScript

Créer un tableau


const fruits = ["pomme", "banane", "orange"];

Transformer une chaîne de caractères en tableau — split()


const phrase = "bonjour tout le monde";

const mots = phrase.split(" ");

console.log(mots);

// ["bonjour", "tout", "le", "monde"]

split(" ") coupe la chaîne à chaque espace.

Transformer un tableau en texte — join()


const mots = ["bonjour", "tout", "le", "monde"];

const phrase = mots.join(" ");

console.log(phrase);

// "bonjour tout le monde"

Ajouter un élément à la fin — push()


const fruits = ["pomme", "banane"];

fruits.push("orange");

console.log(fruits);

// ["pomme", "banane", "orange"]

Supprimer le dernier élément — pop()


const fruits = ["pomme", "banane", "orange"];

fruits.pop();

console.log(fruits);

// ["pomme", "banane"]

Ajouter un élément au début — unshift()


const fruits = ["banane", "orange"];

fruits.unshift("pomme");

console.log(fruits);

// ["pomme", "banane", "orange"]

Supprimer le premier élément — shift()


const fruits = ["pomme", "banane", "orange"];

fruits.shift();

console.log(fruits);

// ["banane", "orange"]

Connaître le nombre d'éléments — length


const fruits = ["pomme", "banane", "orange"];

console.log(fruits.length);

// 3

Récupérer un élément


const fruits = ["pomme", "banane", "orange"];

console.log(fruits[0]);
// "pomme"

console.log(fruits[1]);
// "banane"

Les tableaux commencent à l'index 0.

Parcourir un tableau — for...of


const fruits = ["pomme", "banane", "orange"];

for (const fruit of fruits) {
  console.log(fruit);
}

Vérifier si une valeur existe — includes()


const fruits = ["pomme", "banane", "orange"];

console.log(fruits.includes("banane"));
// true

console.log(fruits.includes("fraise"));
// false

Trouver la position d'un élément — indexOf()


const fruits = ["pomme", "banane", "orange"];

console.log(fruits.indexOf("banane"));

// 1

Extraire une partie du tableau — slice()


const fruits = ["pomme", "banane", "orange", "fraise"];

const selection = fruits.slice(1, 3);

console.log(selection);

// ["banane", "orange"]

Supprimer ou ajouter des éléments — splice()


const fruits = ["pomme", "banane", "orange"];

fruits.splice(1, 1);

console.log(fruits);

// ["pomme", "orange"]

Ici : 1 = position de départ, 1 = nombre d'éléments à supprimer.

Créer un nouveau tableau avec map()


const nombres = [1, 2, 3];

const doubles = nombres.map(function (nombre) {
  return nombre * 2;
});

console.log(doubles);

// [2, 4, 6]

Filtrer un tableau — filter()


const nombres = [1, 2, 3, 4, 5];

const grands = nombres.filter(function (nombre) {
  return nombre > 3;
});

console.log(grands);

// [4, 5]

Trouver un élément — find()


const nombres = [2, 5, 8, 12];

const resultat = nombres.find(function (nombre) {
  return nombre > 6;
});

console.log(resultat);

// 8

Rechercher les voyelles


            
function searchVoyelle(mot) {
    let voyellesFind = [];
    let compteur = 0;

    for (const lettre of mot) {
        if (
            lettre === "a" ||
            lettre === "e" ||
            lettre === "i" ||
            lettre === "o" ||
            lettre === "u" ||
            lettre === "y"
        ) {
            compteur++;
            voyellesFind.push(lettre);
        }
    }

    return {
    compteur: compteur,
    voyelles: voyellesFind
    };
}

Trouver le mot le plus long


function motLePlusLong(phrase) {
  const mots = phrase.split(" ");

  let length = 0;
  let result = "";

  for (const mot of mots) {
    if (mot.length > length) {
      length = mot.length;
      result = mot;
    }
  }

  return result;
}
